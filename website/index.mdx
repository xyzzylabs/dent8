import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="First fact in two minutes" icon="rocket">
    `cargo install dent8-cli` (or `pip install dent8` / `npm i dent8`), then
    `dent8 init` and assert through the firewall. No services.
  </Card>
  <Card title="The 30-second proof" icon="approve-check">
    `dent8 eval` runs a 47-case adversarial corpus against the real firewall and
    against modeled Mem0/Zep semantics: five attack axes blocked vs zero.
  </Card>
  <Card title="Ask what-if" icon="magnifier">
    `dent8 whatif` re-folds the same log under a different trust policy —
    "what would we believe if this source were distrusted?" — deterministically,
    with a structural diff.
  </Card>
  <Card title="Provenance all the way down" icon="seti:lock">
    Authority-weighted arbitration, signed source identity, a hash chain, and an
    optional off-host witness. `dent8 replay` shows exactly why a fact is believed.
  </Card>
</CardGrid>
